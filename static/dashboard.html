        function renderSubmissionsTable(submissions) {
            const tbody = document.getElementById("submissions-table-body");
            tbody.innerHTML = "";

            if (submissions.length === 0) {
                tbody.innerHTML = `
                    
                        No submissions received yet.
                    
                `;
                return;
            }

            submissions.forEach(sub => {
                const date = new Date(sub.created_at).toLocaleString();
                
                // Construct a beautiful payload viewer
                let payloadHtml = '
    ';
                for (const [key, value] of Object.entries(sub.payload)) {
                    payloadHtml += `
  • ${escapeHtml(key)}: ${escapeHtml(String(value))}
    • `;
                }
                payloadHtml += '
';

                const country = sub.geo_country || "Unknown";
                const city = sub.geo_city ? `${sub.geo_city}, ` : "";
                const provider = sub.geo_provider ? ` via ${sub.geo_provider}` : "";

                tbody.innerHTML += `
                    
                        ${date}
                        ${payloadHtml}
                        
                            ${escapeHtml(city)}${escapeHtml(country)}
                            
${escapeHtml(provider)}

                        
                    
                `;
            });
        }

        function renderAnalyticsCharts(analytics) {
            // Geolocation top countries bar chart
            const geoContainer = document.getElementById("geo-chart-container");
            geoContainer.innerHTML = "";
            
            // Sort by count
            const sortedGeo = [...analytics.geo_breakdown].sort((a,b) => b.count - a.count).slice(0, 5);
            const maxGeoCount = sortedGeo.length > 0 ? sortedGeo[0].count : 1;

            if (sortedGeo.length === 0) {
                geoContainer.innerHTML = '
No location metrics yet.
';
            } else {
                sortedGeo.forEach(g => {
                    const widthPct = (g.count / maxGeoCount) * 100;
                    geoContainer.innerHTML += `
                        

                            
${escapeHtml(g.country)}

                            

                                

                            

                            
${g.count}

                        

                    `;
                });
            }

            // Widget stats bar chart
            const widgetContainer = document.getElementById("widget-chart-container");
            widgetContainer.innerHTML = "";

            const sortedWidgets = [...analytics.widget_stats].sort((a,b) => b.count - a.count).slice(0, 5);
            const maxWidgetCount = sortedWidgets.length > 0 ? sortedWidgets[0].count : 1;

            if (sortedWidgets.length === 0) {
                widgetContainer.innerHTML = '
No widgets created yet.
';
            } else {
                sortedWidgets.forEach(w => {
                    const widthPct = (w.count / maxWidgetCount) * 100;
                    widgetContainer.innerHTML += `
                        

                            
${escapeHtml(w.widget_name)}

                            

                                

                            

                            
${w.count}

                        

                    `;
                });
            }
        }

        function addFormConfigField(name = "", type = "text", label = "", required = false, placeholder = "", optionsStr = "") {
            const container = document.getElementById("modal-fields-list");
            const index = currentFieldCount++;

            const fieldDiv = document.createElement("div");
            fieldDiv.className = "field-row";
            fieldDiv.id = `field-row-${index}`;
            fieldDiv.innerHTML = `

            `;

            // Setup select toggle
            const selectEl = fieldDiv.querySelector(".field-key-type");
            const optsDiv = fieldDiv.querySelector(".extra-field-settings");
            const placeDiv = fieldDiv.querySelector(".extra-field-placeholder");
            selectEl.addEventListener("change", (e) => {

        // Open Create Modal
        function openCreateWidgetModal() {
            document.getElementById("modal-title").textContent = "Create Widget";
            document.getElementById("widget-form-id").value = "";
            document.getElementById("widget-form-name").value = "";
            document.getElementById("widget-form-type").value = "signup";
            document.getElementById("widget-form-btn-text").value = "Submit";
            document.getElementById("widget-form-title").value = "";
            document.getElementById("widget-form-desc").value = "";
            
            document.getElementById("modal-fields-list").innerHTML = "";
            currentFieldCount = 0;
            
            // Add default fields (name, email)
            addFormConfigField("name", "text", "Name", true, "Your full name");
            addFormConfigField("email", "email", "Email Address", true, "your.email@domain.com");

            document.getElementById("widget-modal").style.display = "flex";
        }

        // Close Modal
        function closeWidgetModal() {
            document.getElementById("widget-modal").style.display = "none";
        }

        // Save / Submit Create/Edit Form
        document.getElementById("widget-form").addEventListener("submit", function(e) {
            e.preventDefault();
            
            const widgetId = document.getElementById("widget-form-id").value;
            const name = document.getElementById("widget-form-name").value.trim();
            const type = document.getElementById("widget-form-type").value;
            const btnText = document.getElementById("widget-form-btn-text").value.trim();
            const title = document.getElementById("widget-form-title").value.trim();
            const desc = document.getElementById("widget-form-desc").value.trim();

            // Collect custom fields
            const fields = [];
            const fieldRows = document.querySelectorAll(".field-row");
            fieldRows.forEach(row => {
                const fName = row.querySelector(".field-key-name").value.trim();
                const fType = row.querySelector(".field-key-type").value;
                const fLabel = row.querySelector(".field-key-label").value.trim();
                const fReq = row.querySelector(".field-key-req").checked;
                
                const fieldObj = {
                    name: fName,
                    type: fType,
                    label: fLabel,
                    required: fReq
                };

                if (fType === "select") {
                    const optsStr = row.querySelector(".field-key-opts").value.trim();
                    fieldObj.options = optsStr.split(",").map(s => s.trim()).filter(s => s !== "");
                } else {
                    const placeStr = row.querySelector(".field-key-place").value.trim();
                    if (placeStr) fieldObj.placeholder = placeStr;
                }

                fields.push(fieldObj);
            });

            const payload = {
                name,
                type,
                title,
                description: desc,
                button_text: btnText,
                fields,
                display_settings: {}
            };

            const endpoint = widgetId ? `/api/widgets/${widgetId}` : "/api/widgets";
            const method = widgetId ? "PUT" : "POST";

            fetch(endpoint, {
                method: method,
                headers: {
                    "Content-Type": "application/json",
                    "Authorization": `Bearer ${token}`
                },
                body: JSON.stringify(payload)
            })
            .then(res => {
                if (!res.ok) throw new Error("Saving widget failed.");
                return res.json();
            })
            .then(widget => {
                closeWidgetModal();
                refreshDashboardData();
            })
            .catch(err => {
                alert(err.message);
            });
        });

        // Edit widget trigger
        function editWidget(widgetId) {
            fetch(`/api/widgets/${widgetId}`, {
                headers: { "Authorization": `Bearer ${token}` }
            })
            .then(res => res.json())
            .then(widget => {
                document.getElementById("modal-title").textContent = "Edit Widget";
                document.getElementById("widget-form-id").value = widget.id;
                document.getElementById("widget-form-name").value = widget.name;
                document.getElementById("widget-form-type").value = widget.type;
                document.getElementById("widget-form-btn-text").value = widget.button_text;
                document.getElementById("widget-form-title").value = widget.title || "";
                document.getElementById("widget-form-desc").value = widget.description || "";
                
                document.getElementById("modal-fields-list").innerHTML = "";
                currentFieldCount = 0;

                widget.fields.forEach(field => {
                    const optsStr = field.options ? field.options.join(", ") : "";
                    const placeholder = field.placeholder || "";
                    addFormConfigField(field.name, field.type, field.label, field.required, placeholder, optsStr);
                });

                document.getElementById("widget-modal").style.display = "flex";
            });
        }

        // Delete widget trigger
        function deleteWidget(widgetId) {
            if (!confirm("Are you sure you want to permanently delete this widget? All submissions associated with it will be lost.")) {
                return;
            }

            fetch(`/api/widgets/${widgetId}`, {
                method: "DELETE",
                headers: { "Authorization": `Bearer ${token}` }
            })
            .then(res => {
                if (!res.ok) throw new Error("Deleting widget failed.");
                if (activeWidgetId === widgetId) {
                    activeWidgetId = null;
                }
                refreshDashboardData();
            })
            .catch(err => alert(err.message));
        }

        // Safe HTML rendering escape
        function escapeHtml(str) {
            if (!str) return "";
            return str
                .replace(/&/g, "&")
                .replace(//g, ">")
                .replace(/"/g, """)
                .replace(/'/g, "'");
        }
